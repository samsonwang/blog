
#+BEGIN_SRC sh
# 构建一次
docker compose -f docker/docker-compose.yml build     # 或 cd docker 后 docker compose build

# 方式一：一次性编译，跑完即走（最常用）
docker compose -f docker/docker-compose.yml run --rm site nikola build

# 方式二：常驻容器，进去随便折腾
docker compose -f docker/docker-compose.yml up -d
docker compose -f docker/docker-compose.yml exec site bash

# 本地预览（容器内监听 0.0.0.0，宿主机用 localhost:8000）
docker compose -f docker/docker-compose.yml exec site nikola serve -b 0.0.0.0
#+END_SRC
